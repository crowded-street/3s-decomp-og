cmake_minimum_required(VERSION 3.21)
project(sfiii3_decomp LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
file(GLOB_RECURSE DECOMP_SOURCES CONFIGURE_DEPENDS "src/*.c")
add_library(sfiii3_decomp STATIC ${DECOMP_SOURCES})
target_include_directories(sfiii3_decomp PUBLIC src)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(sfiii3_decomp PRIVATE -Wall -Wextra -Werror)
endif()
